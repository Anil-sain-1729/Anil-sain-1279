<script>

let board = ["","","","","","","","",""];

let currentPlayer = "X";

let gameOver = false;

let mode = "human";

let difficulty = "easy";

let playerSymbol = "X";

let robotSymbol = "O";

let history = [];

let playerScore = 0;

let opponentScore = 0;


const cells = document.querySelectorAll(".cell");

const statusText = document.getElementById("status");

const winningLine = document.getElementById("winningLine");


const winningCombinations = [

    [0,1,2],
    [3,4,5],
    [6,7,8],

    [0,3,6],
    [1,4,7],
    [2,5,8],

    [0,4,8],
    [2,4,6]

cells.forEach(cell => {

    cell.addEventListener("click", () => {

        const index = Number(cell.dataset.index);

        makeMove(index);

    });

function setMode(selectedMode) {

    mode = selectedMode;

    document.getElementById("humanBtn")
        .classList.remove("active");

    document.getElementById("robotBtn")
        .classList.remove("active");

    if (mode === "human") {

        document.getElementById("humanBtn")
            .classList.add("active");

        document.getElementById("opponentIcon")
            .textContent = "👤";

    } else {

        document.getElementById("robotBtn")
            .classList.add("active");

        document.getElementById("opponentIcon")
            .textContent = "🤖";

        document.getElementById("opponentName")
            .textContent = "Robot";

        document.getElementById("difficultyBox")
            .style.display = "block";

    }

    resetScore();

    newGame();
}


function chooseSymbol(symbol) {

    playerSymbol = symbol;

    robotSymbol = symbol === "X" ? "O" : "X";

    document.getElementById("xBtn")
        .classList.remove("active");

    document.getElementById("oBtn")
        .classList.remove("active");

    if (symbol === "X") {

        document.getElementById("xBtn")
            .classList.add("active");

    } else {

        document.getElementById("oBtn")
            .classList.add("active");

    }

    newGame();
}


function setDifficulty(level) {

    difficulty = level;

    document.querySelectorAll(".level-btn")
        .forEach(btn => btn.classList.remove("active"));

    document.getElementById(level + "Btn")
        .classList.add("active");

    newGame();
}


function newGame() {

    board = ["","","","","","","","",""];

    history = [];

    gameOver = false;

    winningLine.style.display = "none";

    currentPlayer = "X";

    cells.forEach(cell => {

        cell.textContent = "";

        cell.classList.remove("x","o");

    });

    updateStatus();

    if (mode === "robot" &&
        currentPlayer === robotSymbol) {

        setTimeout(robotMove, 500);

    }

}


function makeMove(index) {

    if (gameOver) return;

    if (board[index] !== "") return;

    if (mode === "robot" &&
        currentPlayer === robotSymbol) return;

    placeMove(index, currentPlayer);

}


function placeMove(index, symbol) {

    history.push([...board]);

    board[index] = symbol;

    const cell = cells[index];

    cell.textContent = symbol === "X" ? "×" : "○";

    cell.classList.add(
        symbol === "X" ? "x" : "o"
    );

    const result = checkWinner();

    if (result) {

        finishGame(result);

        return;

    }

    if (!board.includes("")) {

        gameOver = true;

        statusText.textContent = "🤝 It's a Draw!";

        return;

    }

    currentPlayer =
        currentPlayer === "X" ? "O" : "X";

    updateStatus();

    if (mode === "robot" &&
        currentPlayer === robotSymbol &&
        !gameOver) {

        setTimeout(robotMove, 500);

    }

}


function updateStatus() {

    if (mode === "human") {

        statusText.textContent =
            "Player " + currentPlayer + "'s Turn";

    } else {

        if (currentPlayer === playerSymbol) {

            statusText.textContent =
                "🎯 Your Turn (" +
                playerSymbol +
                ")";

        } else {

            statusText.textContent =
                "🤖 Robot's Turn (" +
                robotSymbol +
                ")";

        }

    }

}


function checkWinner() {

    for (let combination of winningCombinations) {

        const [a,b,c] = combination;

        if (
            board[a] &&
            board[a] === board[b] &&
            board[a] === board[c]
        ) {

            return {
                winner: board[a],
                combination: combination
            };

        }

    }

    return null;

}


function finishGame(result) {

    gameOver = true;

    drawWinningLine(result.combination);

    if (mode === "human") {

        if (result.winner === "X") {

            playerScore++;

        } else {

            opponentScore++;

        }

    } else {

        if (result.winner === playerSymbol) {

            playerScore++;

            statusText.textContent =
                "🎉 You Win!";

        } else {

            opponentScore++;

            statusText.textContent =
                "🤖 Robot Wins!";

        }

    }

    updateScore();

    if (mode === "human") {

        statusText.textContent =
            "🏆 Player " +
            result.winner +
            " Wins!";

    }

}


function updateScore() {

    document.getElementById("playerScore")
        .textContent = playerScore;

    document.getElementById("opponentScore")
        .textContent = opponentScore;

}


function drawWinningLine(combination) {

    const first = cells[combination[0]];

    const last = cells[combination[2]];

    const boardRect =
        document.getElementById("board")
            .getBoundingClientRect();

    const firstRect =
        first.getBoundingClientRect();

    const lastRect =
        last.getBoundingClientRect();

    const x1 =
        firstRect.left +
        firstRect.width / 2 -
        boardRect.left;

    const y1 =
        firstRect.top +
        firstRect.height / 2 -
        boardRect.top;

    const x2 =
        lastRect.left +
        lastRect.width / 2 -
        boardRect.left;

    const y2 =
        lastRect.top +
        lastRect.height / 2 -
        boardRect.top;

    const length =
        Math.sqrt(
            Math.pow(x2-x1,2) +
            Math.pow(y2-y1,2)
        );

    const angle =
        Math.atan2(y2-y1, x2-x1) *
        180 / Math.PI;

    winningLine.style.width =
        length + "px";

    winningLine.style.left =
        x1 + "px";

    winningLine.style.top =
        y1 + "px";

    winningLine.style.transform =
        "rotate(" + angle + "deg)";

    winningLine.style.display = "block";

}


function undoMove() {

    if (gameOver) return;

    if (history.length === 0) return;

    if (mode === "robot") {

        if (history.length >= 2) {

            board = history[history.length - 2];

            history.splice(-2);

        } else {

            board = history.pop();

        }

    } else {

        board = history.pop();

    }

    redrawBoard();

    currentPlayer = "X";

    if (mode === "human" &&
        history.length % 2 !== 0) {

        currentPlayer = "O";

    }

    updateStatus();

}


function redrawBoard() {

    cells.forEach((cell,index) => {

        cell.textContent =
            board[index] === "X"
                ? "×"
                : board[index] === "O"
                    ? "○"
                    : "";

        cell.classList.remove("x","o");

        if (board[index]) {

            cell.classList.add(
                board[index] === "X"
                    ? "x"
                    : "o"
            );

        }

    });

}


function resetScore() {

    playerScore = 0;

    opponentScore = 0;

    updateScore();

    newGame();

}

function robotMove() {

    if (gameOver) return;

    if (currentPlayer !== robotSymbol) return;

    let index;

    if (difficulty === "easy") {

        index = randomMove();

    }

    else if (difficulty === "medium") {

        index = mediumMove();

    }

    else {

        index = bestMove();

    }

    if (index !== -1) {

        placeMove(index, robotSymbol);

    }

}


function randomMove() {

    const empty = [];

    board.forEach((value,index) => {

        if (value === "") {

            empty.push(index);

        }

    });

    if (empty.length === 0) return -1;

    return empty[
        Math.floor(Math.random() * empty.length)
    ];

}


function mediumMove() {

    /* First try to win */

    for (let i=0; i<9; i++) {

        if (board[i] === "") {

            board[i] = robotSymbol;

            if (checkWinner()) {

                board[i] = "";

                return i;

            }

            board[i] = "";

        }

    }

    /* Then block player */

    for (let i=0; i<9; i++) {

        if (board[i] === "") {

            board[i] = playerSymbol;

            if (checkWinner()) {

                board[i] = "";

                return i;

            }

            board[i] = "";

        }

    }

    /* Center */

    if (board[4] === "") return 4;

    /* Random */

    return randomMove();

}


function bestMove() {

    let bestScore = -Infinity;

    let move = -1;

    for (let i=0; i<9; i++) {

        if (board[i] === "") {

            board[i] = robotSymbol;

            let score =
                minimax(
                    board,
                    false
                );

            board[i] = "";

            if (score > bestScore) {

                bestScore = score;

                move = i;

            }

        }

    }

    return move;

}


function minimax(state, maximizing) {

    const result = checkWinner();

    if (result) {

        if (result.winner === robotSymbol) {

            return 10;

        }

        if (result.winner === playerSymbol) {

            return -10;

        }

    }

    if (!state.includes("")) {

        return 0;

    }

    if (maximizing) {

        let best = -Infinity;

        for (let i=0; i<9; i++) {

            if (state[i] === "") {

                state[i] = robotSymbol;

                best = Math.max(
                    best,
                    minimax(state,false)
                );

                state[i] = "";

            }

        }

        return best;

    } else {

        let best = Infinity;

        for (let i=0; i<9; i++) {

            if (state[i] === "") {

                state[i] = playerSymbol;

                best = Math.min(
                    best,
                    minimax(state,true)
                );

                state[i] = "";

            }

        }

        return best;

    }

}
